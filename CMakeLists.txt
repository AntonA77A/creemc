cmake_minimum_required(VERSION 3.20)
project(creemc CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(CREEMC_STATIC "Komplett statisch linken (Linux, z.B. Alpine/musl)" OFF)

file(GLOB SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.cpp)

# ---- Eingebettete Dateien: Versionsdaten, Spielregeln, Java-Bridge ------------
set(DATA_DIR ${CMAKE_SOURCE_DIR}/data)
file(GLOB VERSION_FILES ${DATA_DIR}/versions/*.json)
set(EMBED_NAMES "game.json")
set(EMBED_PATHS "${DATA_DIR}/game.json")
foreach(f ${VERSION_FILES})
  get_filename_component(stem ${f} NAME_WE)
  list(APPEND EMBED_NAMES "versions/${stem}.json")
  list(APPEND EMBED_PATHS "${f}")
endforeach()
if(EXISTS ${CMAKE_SOURCE_DIR}/bridge/creemc-bridge.jar)
  list(APPEND EMBED_NAMES "bridge.jar")
  list(APPEND EMBED_PATHS "${CMAKE_SOURCE_DIR}/bridge/creemc-bridge.jar")
endif()

if(WIN32)
  # Windows: als RCDATA-Ressourcen ("versions/777.json" -> VERSIONS_777_JSON)
  set(RC_LINES "")
  foreach(n p IN ZIP_LISTS EMBED_NAMES EMBED_PATHS)
    string(TOUPPER "${n}" id)
    string(REGEX REPLACE "[/.]" "_" id "${id}")
    string(APPEND RC_LINES "${id} RCDATA \"${p}\"\n")
  endforeach()
  file(WRITE ${CMAKE_BINARY_DIR}/data.rc "${RC_LINES}")
  list(APPEND SOURCES ${CMAKE_BINARY_DIR}/data.rc)
else()
  # Linux: per .incbin direkt ins Programm
  enable_language(ASM)
  set(ASM_LINES ".section .rodata\n")
  set(TABLE "struct EmbeddedFile { const char* name; const char* start; const char* end; };\n")
  set(ENTRIES "")
  set(i 0)
  foreach(n p IN ZIP_LISTS EMBED_NAMES EMBED_PATHS)
    string(APPEND ASM_LINES ".global creemc_res_${i}_start\n.global creemc_res_${i}_end\n.balign 16\ncreemc_res_${i}_start:\n.incbin \"${p}\"\ncreemc_res_${i}_end:\n.byte 0\n")
    string(APPEND TABLE "extern \"C\" const char creemc_res_${i}_start[], creemc_res_${i}_end[];\n")
    string(APPEND ENTRIES "  {\"${n}\", creemc_res_${i}_start, creemc_res_${i}_end},\n")
    math(EXPR i "${i} + 1")
  endforeach()
  string(APPEND ASM_LINES ".section .note.GNU-stack,\"\",@progbits\n")
  string(APPEND TABLE "extern const EmbeddedFile EMBEDDED_FILES[] = {\n${ENTRIES}};\nextern const int EMBEDDED_COUNT = ${i};\n")
  file(WRITE ${CMAKE_BINARY_DIR}/embed.S "${ASM_LINES}")
  file(WRITE ${CMAKE_BINARY_DIR}/embedded.cpp "${TABLE}")
  list(APPEND SOURCES ${CMAKE_BINARY_DIR}/embed.S ${CMAKE_BINARY_DIR}/embedded.cpp)
  set_source_files_properties(${CMAKE_BINARY_DIR}/embed.S PROPERTIES OBJECT_DEPENDS "${EMBED_PATHS}")
endif()

add_executable(creemc ${SOURCES})

if(MSVC)
  target_compile_options(creemc PRIVATE /utf-8 /W3 /EHsc /permissive-)
  target_compile_definitions(creemc PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS)
  set_property(TARGET creemc PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")   # keine VC++-DLLs nötig
  target_link_libraries(creemc PRIVATE ws2_32 bcrypt winhttp)
else()
  target_compile_options(creemc PRIVATE -O2 -Wall)
  if(CREEMC_STATIC)
    set(OPENSSL_USE_STATIC_LIBS TRUE)
    target_link_options(creemc PRIVATE -static)
  endif()
  find_package(OpenSSL REQUIRED)
  find_package(Threads REQUIRED)
  target_link_libraries(creemc PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
endif()
